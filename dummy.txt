cd /tmp

# Download
wget https://www.python.org/ftp/python/3.9.18/Python-3.9.18.tgz

# Extract
tar -xf Python-3.9.18.tgz
cd Python-3.9.18

# Configure
./configure --enable-optimizations --prefix=/usr/local/python3.9 --with-ensurepip=install

# Build (use all available cores)
make -j$(nproc)

# Install
sudo make altinstall

# Add Python 3.9 to PATH permanently
echo 'export PATH=/usr/local/python3.9/bin:$PATH' >> ~/.bashrc
source ~/.bashrc

# Verify installation
python3.9 --version

# Upgrade pip
python3.9 -m pip install --upgrade pip

# ── VENV SETUP ──────────────────────────────────────

# Install venv module
python3.9 -m pip install virtualenv

# Create virtual environment (change 'myenv' to your preferred name)
python3.9 -m venv ~/myenv

# Activate the virtual environment
source ~/myenv/bin/activate

# Verify venv is using Python 3.9.18
python --version
pip --version

# Deactivate when done
# deactivate


# Install dependencies
sudo apt install -y libopenblas-dev libblas-dev m4 cmake cython python3-dev

# Install Python build deps
pip install numpy pyyaml setuptools cffi typing_extensions

# Clone PyTorch
cd ~
git clone --recursive https://github.com/pytorch/pytorch
cd pytorch

# Set build flags for Pi 32-bit
export USE_CUDA=0
export USE_MKLDNN=0
export USE_QNNPACK=0
export USE_NNPACK=0
export USE_DISTRIBUTED=0
export MAX_JOBS=2

# Build and install into venv
python setup.py install
